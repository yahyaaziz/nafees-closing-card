/** * Nafees Closing Card — backend v4 * Paste into a Google Sheet's Apps Script editor (Extensions > Apps Script), replacing * any previous version. Redeploy (Deploy > Manage deployments > Edit > New version) * after pasting — the URL stays the same, no need to update the frontend again. * IMPORTANT: after redeploying, double check "Who has access" is still set to "Anyone" — * this has silently reset before and caused every request to fail without an error shown. * * v4 change: added a 'Customer History' column (longstanding vs. new customer * classification), editable only from the Card Log — never a live Closing Card field. * Deliberately owner/manager-filled, not staff-filled. * * v3 change: doPost is wrapped so any internal error gets written directly into * the sheet as a visible row, instead of failing silently. The app talks to this * script in 'no-cors' mode, which hides all errors from the browser — so a crash * inside doPost would otherwise leave zero trace anywhere. * * Supports three actions from the app: * 'start' — fired the instant a Closing Card is opened. Guarantees every visit * is counted even if the card is abandoned partway through. * 'complete' — the finished card. Fills in the row created by 'start', or appends * a new one if the start ping never arrived (e.g. was offline). * 'update' — edits an existing row by Visit ID. Used by the in-app Card Log to * fill in a name/phone/customer-history learned after the fact. * A GET request (used by the in-app Card Log) returns the most recent 50 rows as JSON. */ var HEADERS = [ 'Visit ID','Status','Started At','Completed At','Employee','Gender', 'Competitor Mentioned','Referral Mentioned', 'Consideration Item 1','Consideration Item 2','Consideration Item 3', 'Group Composition','Refreshment', 'Variant Signal','Variant Type','Weight Direction', 'Price Signal','Pushback Reason','Outcome', 'Contact Shared','Contact Name','Contact Phone','Customer History' ]; function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('ClosingCards'); if (!sheet) sheet = ss.insertSheet('ClosingCards'); if (sheet.getLastRow() === 0) sheet.appendRow(HEADERS); return sheet; } function findRowByVisitId_(sheet, visitId) { var lastRow = sheet.getLastRow(); if (lastRow < 2) return -1; var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { if (ids[i][0] === visitId) return i + 2; } return -1; } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function formatItem_(item) { if (!item) return ''; return item.weightBand ? (item.label + ' — ' + item.weightBand) : item.label; } // Writes a visible error row so failures never go unnoticed again, even under 'no-cors'. function logError_(context, message) { try { var sheet = getSheet_(); sheet.appendRow(['ERROR', context, new Date(), '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', String(message).slice(0, 500), '']); } catch (e2) { // if even error logging fails, there's nothing further we can do here } } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { logError_('doPost', 'Request arrived with no postData — the POST body was empty or missing.'); return jsonOut_({status: 'error', message: 'no postData'}); } var data; try { data = JSON.parse(e.postData.contents); } catch (parseErr) { logError_('doPost JSON.parse', 'Body was not valid JSON: ' + e.postData.contents.slice(0, 200)); return jsonOut_({status: 'error', message: 'invalid JSON'}); } var sheet = getSheet_(); var action = data.action || 'complete'; if (action === 'start') { var blankRow = [data.visitId, 'started', new Date(), '', data.employee || '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '']; sheet.appendRow(blankRow); return jsonOut_({status: 'ok'}); } var rowIndex = findRowByVisitId_(sheet, data.visitId); if (action === 'complete') { var items = data.considerations || []; var rowValues = [ data.visitId, 'completed', '', new Date(), data.employee || '', data.gender || '', data.competitorMentioned ? 'Yes' : 'No', data.referralMentioned ? 'Yes' : 'No', formatItem_(items[0]), formatItem_(items[1]), formatItem_(items[2]), data.groupComposition || '', data.refreshment || '', data.variant || '', data.variantType || '', data.weightDirection || '', data.price || '', data.pushbackReason || '', data.outcome || '', data.contactShared === true ? 'Yes' : (data.contactShared === false ? 'No' : ''), data.contactName || '', data.contactPhone || '', data.customerHistory || '' ]; if (rowIndex > -1) { var startedAt = sheet.getRange(rowIndex, 3).getValue(); sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]); sheet.getRange(rowIndex, 3).setValue(startedAt); // preserve the original start timestamp } else { sheet.appendRow(rowValues); // start ping never arrived — append fresh } return jsonOut_({status: 'ok'}); } if (action === 'update') { if (rowIndex === -1) return jsonOut_({status: 'error', message: 'visit not found'}); if (data.contactName !== undefined) sheet.getRange(rowIndex, 21).setValue(data.contactName); if (data.contactPhone !== undefined) sheet.getRange(rowIndex, 22).setValue(data.contactPhone); if (data.outcome !== undefined) sheet.getRange(rowIndex, 19).setValue(data.outcome); if (data.customerHistory !== undefined) sheet.getRange(rowIndex, 23).setValue(data.customerHistory); return jsonOut_({status: 'ok'}); } return jsonOut_({status: 'error', message: 'unknown action'}); } catch (err) { logError_('doPost top-level', err && err.message ? err.message : String(err)); return jsonOut_({status: 'error', message: 'internal error, logged to sheet'}); } } function doGet(e) { try { var sheet = getSheet_(); // Diagnostic-only path: a GET carrying ?ping=1 performs the same kind of write // a POST normally would. This exists purely to isolate whether POST specifically // is being blocked somewhere between the app and Google, versus GET working fine. if (e && e.parameter && e.parameter.ping === '1') { sheet.appendRow([e.parameter.visitId || 'ping', 'started (via GET test)', new Date(), '', e.parameter.employee || 'GET ping', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '', '']); return jsonOut_({status: 'ok', via: 'GET'}); } var lastRow = sheet.getLastRow(); if (lastRow < 2) return jsonOut_({cards: []}); var startRow = Math.max(2, lastRow - 49); // last 50 rows var numRows = lastRow - startRow + 1; var values = sheet.getRange(startRow, 1, numRows, HEADERS.length).getValues(); var cards = values.map(function (row) { var obj = {}; HEADERS.forEach(function (h, i) { obj[h] = row[i]; }); return obj; }).reverse(); // most recent first return jsonOut_({cards: cards}); } catch (err) { logError_('doGet', err && err.message ? err.message : String(err)); return jsonOut_({cards: [], error: String(err)}); } }
